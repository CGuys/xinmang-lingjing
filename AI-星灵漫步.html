// 微信小程序原生 Canvas 2D 导出海报完整示范
Page({
  data: {
    isFlipped: false,
    cardData: {
      roman: "XVII",
      name: "星星 (The Star)",
      keyword: "内在治愈 · 灵感觉醒",
      insight: "过往困顿的迷雾正在消散。当下的你正在经历精神蜕变，请彻底信任你的内在直觉。",
      affirmation: "你无需成为任何人的解药，你本身就是一片自由旷野。"
    }
  },

  // 翻牌交互
  onFlipCard() {
    this.setData({ isFlipped: !this.data.isFlipped });
  },

  // 生成海报入口
  async generatePoster() {
    wx.showLoading({ title: '正在绘制灵性能量卡...' });

    const query = wx.createSelectorQuery();
    query.select('#posterCanvas')
      .fields({ node: true, size: true })
      .exec(async (res) => {
        if (!res[0] || !res[0].node) {
          wx.hideLoading();
          return;
        }

        const canvas = res[0].node;
        const ctx = canvas.getContext('2d');
        const dpr = wx.getSystemInfoSync().pixelRatio || 2;

        // 设置真实物理像素抗锯齿 (750 x 1334 标准海报尺寸)
        const width = 750;
        const height = 1334;
        canvas.width = width * dpr;
        canvas.height = height * dpr;
        ctx.scale(dpr, dpr);

        // 1. 绘制星空暗调渐变背景
        const bgGrad = ctx.createLinearGradient(0, 0, 0, height);
        bgGrad.addColorStop(0, '#100e24');
        bgGrad.addColorStop(0.5, '#070611');
        bgGrad.addColorStop(1, '#020108');
        ctx.fillStyle = bgGrad;
        ctx.fillRect(0, 0, width, height);

        // 2. 绘制金色流光边框
        ctx.strokeStyle = 'rgba(212, 175, 55, 0.4)';
        ctx.lineWidth = 4;
        ctx.strokeRect(36, 36, width - 72, height - 72);

        // 3. 绘制卡片标题
        ctx.fillStyle = '#fef08a';
        ctx.font = 'bold 36px sans-serif';
        ctx.textAlign = 'center';
        ctx.fillText("『 今日心灵投射报告 』", width / 2, 130);

        // 4. 绘制牌面核心能量词
        ctx.fillStyle = '#ffffff';
        ctx.font = 'bold 48px sans-serif';
        ctx.fillText(this.data.cardData.name, width / 2, 230);

        ctx.fillStyle = '#d4af37';
        ctx.font = '26px sans-serif';
        ctx.fillText(`【 ${this.data.cardData.keyword} 】`, width / 2, 285);

        // 5. 自动换行绘制 AI 洞见正文
        ctx.fillStyle = '#cbd5e1';
        ctx.font = '28px sans-serif';
        ctx.textAlign = 'left';
        this.drawWrappedText(ctx, this.data.cardData.insight, 80, 420, 590, 48);

        // 6. 绘制金句引用框
        ctx.fillStyle = 'rgba(212, 175, 55, 0.08)';
        ctx.fillRect(80, 680, 590, 160);
        ctx.strokeStyle = 'rgba(212, 175, 55, 0.3)';
        ctx.strokeRect(80, 680, 590, 160);

        ctx.fillStyle = '#fde68a';
        ctx.font = 'italic 26px sans-serif';
        this.drawWrappedText(ctx, `“${this.data.cardData.affirmation}”`, 110, 740, 530, 42);

        // 7. 绘制底部圆形小程序码与作者引导
        // (注：真实项目通过 wx.getImageInfo 先缓存服务端生成的 SunCode 小程序码临时路径)
        this.drawCircleQRCode(ctx, width / 2, 1060, 80);

        ctx.fillStyle = '#94a3b8';
        ctx.font = '22px sans-serif';
        ctx.textAlign = 'center';
        ctx.fillText("长按识别小程序 · 探索你的深层潜意识", width / 2, 1200);

        // 8. 导出临时文件并唤起保存到相册
        wx.canvasToTempFilePath({
          canvas,
          success: (fileRes) => {
            wx.hideLoading();
            wx.saveImageToPhotosAlbum({
              filePath: fileRes.tempFilePath,
              success: () => wx.showToast({ title: '已保存至相册' })
            });
          }
        });
      });
  },

  // 核心工具：Canvas 文本自动测量折行算法
  drawWrappedText(ctx, text, x, y, maxWidth, lineHeight) {
    let words = text.split('');
    let line = '';
    for (let n = 0; n < words.length; n++) {
      let testLine = line + words[n];
      let metrics = ctx.measureText(testLine);
      if (metrics.width > maxWidth && n > 0) {
        ctx.fillText(line, x, y);
        line = words[n];
        y += lineHeight;
      } else {
        line = testLine;
      }
    }
    ctx.fillText(line, x, y);
  },

  // 绘制圆形小程序码占位
  drawCircleQRCode(ctx, x, y, radius) {
    ctx.save();
    ctx.beginPath();
    ctx.arc(x, y, radius, 0, Math.PI * 2);
    ctx.strokeStyle = '#d4af37';
    ctx.lineWidth = 3;
    ctx.stroke();
    ctx.fillStyle = '#1e1b4b';
    ctx.fill();
    ctx.fillStyle = '#facc15';
    ctx.font = '20px sans-serif';
    ctx.textAlign = 'center';
    ctx.fillText('小程序码', x, y + 8);
    ctx.restore();
  }
});
1. 微信审核“迷信红线”必坑准则

严禁出现“算命、八字合婚、凶吉测算、改运转运”等词汇。必须在小程序所有页面包装为“心理学潜意识投射”、“心灵疗愈卡片”、“每日灵感便签”，并在显著位置注明：“本测试仅供娱乐与心理纾压，不构成任何现实指导建议”。

2. Canvas 2D 导出模糊或白屏解决方案
  • 必须使用 wx.getSystemInfoSync().pixelRatio 给 Canvas 物理尺寸乘上 dpr，再用 ctx.scale(dpr, dpr) 缩放，否则在 Retina 视网膜屏上导出海报字迹必然严重发虚。
  • 外部网络图片（如用户微信头像、动态小程序码）绘制前，必须先经过 wx.downloadFile 或 wx.getImageInfo 下载为本地临时路径，并配置微信公众平台的合法 downloadFile 白名单域名。
3. 激励视频变现与防刷机制

前端监听激励视频 rewardedVideoAd.onClose(res => res.isEnded) 极易被抓包篡改。商业化必须开启微信“激励视频服务端回调 (Server-to-Server)”，由微信官方服务器直连开发者 Go/Node.js 后端加点，彻底杜绝外挂刷算力。
